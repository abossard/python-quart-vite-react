Design Principles
  • Deep modules — simple API, complex internals
  • No provider coupling — host injects LLM factory
  • FSM-validated state — invalid transitions rejected
  • Grokking Simplicity — data / calculations / actions

The LLM Factory Protocol

The host decides which provider to use. Agent Builder never imports one.


agent_builder/llm_protocol.py

@dataclass(frozen=True)
class LLMConfig:
    model: str = ""
    temperature: float = 0.0
    max_tokens: int = 4096
    api_key: str = ""
    base_url: str = ""

class LLMFactory(Protocol):
    def __call__(
        self, config: LLMConfig
    ) -> BaseChatModel: ...

Run Status — Finite State Machine

─ START →
🔄 RUNNING
─ COMPLETE →
✅ COMPLETED

├─ FAIL →
❌ FAILED
├─ TRUNCATE →
⚠️ TRUNCATED

from agent_builder import transition, RunStatus, RunEvent

status = transition(RunStatus.PENDING, RunEvent.START)    # → RUNNING
status = transition(status, RunEvent.COMPLETE)             # → COMPLETED
transition(RunStatus.COMPLETED, RunEvent.START)            # raises InvalidTransition

# Define an agent
agent = workbench.create_agent(AgentDefinitionCreate(
    name="SLA Analyzer",
    system_prompt="Analyze ticket SLA breaches and report findings.",

      inputPrompt: 'Analyze VPN tickets'

    // run.status === 'running'
    // SSE events auto-update runs array
  }

  return runs.map(run => (
    <div key={run.id}>
      {run.status} — {getRunActivity(run.id).length} events
    </div>
  ))
}

Step 1 — integration.py

This file is the only place where your app touches agent_builder internals.

"""integration.py — Wire agent_builder into your app."""
import os
from pathlib import Path
from langchain_openai import ChatOpenAI
from langchain_core.tools import StructuredTool
from agent_builder import (
    WorkbenchService, ChatService, ToolRegistry, LLMConfig,
)

# ── 1. LLM Factory ──────────────────────────────────────────────
def llm_factory(config: LLMConfig):
    """Host decides the provider. Agent Builder never imports one."""
    return ChatOpenAI(
        model=config.model or "gpt-4o-mini",
        api_key=config.api_key or os.getenv("OPENAI_API_KEY"),
        temperature=config.temperature,
        max_tokens=config.max_tokens or None,
    )

# ── 2. Tool Registry ────────────────────────────────────────────
registry = ToolRegistry()

# Register any LangChain StructuredTool
registry.register(StructuredTool.from_function(
    name="get_weather",
    description="Get current weather for a city",
    func=lambda city: f"Sunny, 22°C in {city}",
))

# ── 3. Services ─────────────────────────────────────────────────
workbench_service = WorkbenchService(
    tool_registry=registry,
    llm_factory=llm_factory,
    db_path=Path("data/agents.db"),
    default_model="gpt-4o-mini",
)

chat_service = ChatService(
    tool_registry=registry,
    llm_factory=llm_factory,
)

Step 2 — app.py

Mount the blueprint. That's it — 20+ endpoints are live.

"""app.py — Quart application."""
import asyncio
from quart import Quart
from quart_cors import cors
from agent_builder.routes import agent_builder_bp, configure_blueprint
from agent_builder.engine.event_bus import agent_event_bus
from integration import workbench_service, chat_service

app = Quart(__name__)
app = cors(app)

# Wire services into the blueprint
configure_blueprint(
    workbench_service=workbench_service,
    chat_service=chat_service,
)
app.register_blueprint(agent_builder_bp)

@app.before_serving
async def startup():
    # Capture event loop for SSE thread-safe dispatch
    agent_event_bus.set_loop(asyncio.get_running_loop())

@app.route("/health")
async def health():
    return {"status": "ok", "agents": "enabled"}

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5001)

Pattern: Ollama (Local)

from langchain_ollama import ChatOllama
from agent_builder import LLMConfig

def ollama_factory(config: LLMConfig):
    return ChatOllama(
        model=config.model or "llama3.1",
        base_url=config.base_url or "http://localhost:11434",
        temperature=config.temperature,
    )
Run: ollama serve && ollama pull llama3.1

Pattern: Multi-Provider Router

Switch providers per-agent using LLMConfig.extra

from agent_builder import LLMConfig

def router_factory(config: LLMConfig):
    provider = config.extra.get("provider", "openai")

    if provider == "azure":
        return azure_factory(config)
    elif provider == "ollama":
        return ollama_factory(config)
    else:
        return openai_factory(config)

# Per-agent override via output_schema or extra config:
# agent.model = "llama3.1"
# → factory receives LLMConfig(model="llama3.1")
Tip: Each agent can specify its own model and temperature in the agent definition. The factory receives these via LLMConfig.

import { useAgentManager, useRunManager, parseRunOutput } from 'agent-builder-ui'

function AgentDashboard() {
  const { agents, loading: agentsLoading } = useAgentManager()
  const { runs, startRun } = useRunManager()

  if (agentsLoading) return <p>Loading...</p>

  return (
    <div>
      <h1>My Agents</h1>
      {agents.map(agent => (
        <div key={agent.id} className="agent-card">
          <h3>{agent.name}</h3>
          <p>{agent.description}</p>
          <button onClick={() => startRun(agent.id, {
            inputPrompt: 'Run analysis'
          })}>
            ▶ Run
          </button>
        </div>
      ))}

      <h2>Recent Runs</h2>
      {runs.map(run => (
        <div key={run.id}>
          {run.status === 'running' ? '🔄' : '✅'} {run.status}
          <pre>{JSON.stringify(parseRunOutput(run.output), null, 2)}</pre>
        </div>
      ))}
    </div>
  )
}

Live Activity Monitor

import {
  subscribeSSE, SSE_STATE,
  eventSummary, formatTime, shortId
} from 'agent-builder-ui'
import { useEffect, useState } from 'react'

function ActivityFeed() {
  const [events, setEvents] = useState([])
  const [connected, setConnected] = useState(false)

  useEffect(() => {
    return subscribeSSE({
      onEvent: (evt) => setEvents(prev => [...prev.slice(-99), evt]),
      onStateChange: (s) => setConnected(s === SSE_STATE.CONNECTED),
    })
  }, [])

  return (
    <div>
      <span>{connected ? '🟢' : '🔴'} SSE</span>
      {events.map((evt, i) => (
        <div key={i}>
          <code>{formatTime(evt.timestamp)}</code>
          <span>{shortId(evt.runId)}</span>
          <span>{eventSummary(evt)}</span>
        </div>
      ))}
    </div>
  )
}

Agent Create Form

import { useAgentManager, buildModelOptions } from 'agent-builder-ui'
import { useState } from 'react'

function CreateAgentForm() {
  const { tools, uiConfig, createAgent } = useAgentManager()
  const [name, setName] = useState('')
  const [prompt, setPrompt] = useState('')
  const [selectedTools, setSelectedTools] = useState([])

  const models = buildModelOptions(
    uiConfig?.llm_catalog?.available_models || [], ''
  )

  const handleSubmit = async () => {
    await createAgent({
      name,
      system_prompt: prompt,
      tool_names: selectedTools,
    })
    setName(''); setPrompt(''); setSelectedTools([])
  }

  return (
    <form onSubmit={e => { e.preventDefault(); handleSubmit() }}>
      <input value={name} onChange={e => setName(e.target.value)}
             placeholder="Agent name" />
      <textarea value={prompt} onChange={e => setPrompt(e.target.value)}
                placeholder="System prompt" />

      <h4>Tools ({tools.length} available)</h4>
      {tools.map(t => (
        <label key={t.name}>
          <input type="checkbox"
                 checked={selectedTools.includes(t.name)}
                 onChange={() => setSelectedTools(prev =>
                   prev.includes(t.name)
                     ? prev.filter(n => n !== t.name)
                     : [...prev, t.name]
                 )} />
          {t.name}
        </label>
      ))}

      <button type="submit">Create Agent</button>
    </form>
  )
}
